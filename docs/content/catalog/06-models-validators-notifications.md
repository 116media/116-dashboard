## Validators

**Path:** `src/modules/catalog/presentation/utils/validators/`

### `catalog.categories.validator.ts`

```ts
export const CategoriesValidator = {
    contentTypeId: (label: string) => [required(label)],
    name: (label: string) => [required(label), max(label, 100)],
    description: (label: string) => [max(label, 500)],
    isFree: (label: string) => [required(label)],
    pricingTierId: (label: string) => [required(label)],
    priceUsd: (label: string) => [required(label), min(label, 0)],
};
```

### `catalog.customers.validator.ts`

```ts
export const CustomersValidator = {
    fullName: (label: string) => [required(label), max(label, 100)],
    email: (label: string) => [required(label), email(label)],
    phone: (label: string) => [max(label, 30)],
    company: (label: string) => [max(label, 100)],
    notes: (label: string) => [max(label, 500)],
};
```

### `catalog.packages.validator.ts`

```ts
export const PackagesValidator = {
    name: (label: string) => [required(label), max(label, 100)],
    description: (label: string) => [max(label, 500)],
    flatPriceUsd: (label: string) => [required(label), min(label, 0)],
    categoryId: (label: string) => [required(label)],
    isRequired: (label: string) => [required(label)],
    quantity: (label: string) => [required(label), min(label, 1)],
};
```
